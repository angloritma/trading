//@version=6
strategy(
     title = "Angloritma Buy/Sell signal - Quant, EMA Cross, TP 2% & SL Close 2%",
     shorttitle = "Angloritma Quant + EMA (TP/SL 5%)",
     overlay = true,
     initial_capital = 100000000,
     currency = currency.IDR,
     default_qty_type = strategy.percent_of_equity,
     default_qty_value = 100,
     pyramiding = 0,
     commission_type = strategy.commission.percent,
     commission_value = 0.1,
     process_orders_on_close = true,
     calc_on_order_fills = true
     )

// ============================================================
// INPUT PARAMETER
// ============================================================

entryMode     = input.string("Fleksibel (Konfirmasi)", "Mode Entry BUY", 
                             options = ["Fleksibel (Konfirmasi)", "Ketat (Candle Bersamaan)"], 
                             group = "Entry Logic")

emaLength     = input.int(13, "EMA Short Length", group = "Moving Averages")
emaLongLength = input.int(21, "EMA Long Length", group = "Moving Averages")

takeProfitPct = input.float(5.0, "Take Profit Running Price (%)", group = "Risk Management")
stopLossPct   = input.float(5.0, "Stop Loss Candle Close (%)", group = "Risk Management")

startDate = input.time(timestamp("2020-01-01T00:00:00"), "Start Date", group = "Backtest Period")
endDate   = input.time(timestamp("2030-12-31T23:59:59"), "End Date", group = "Backtest Period")

inDateRange = time >= startDate and time <= endDate

// ============================================================
// INDIKATOR: EMA & QUANT FORECASTING
// ============================================================

ema13 = ta.ema(close, emaLength)
ema21 = ta.ema(close, emaLongLength)

bullishCross = ta.crossover(ema13, ema21)
bearishCross = ta.crossunder(ema13, ema21)

int length_period = timeframe.isdaily ? 30 : 60
quantSma = ta.sma(close, length_period)

// Logika status warna garis Quant
bool isQuantBlue     = close >= quantSma
bool quantTurnedBlue = isQuantBlue and not isQuantBlue[1] // Baru berubah ke biru

// ============================================================
// LOGIKA SINYAL BUY
// ============================================================

bool buyCondition = false

if entryMode == "Ketat (Candle Bersamaan)"
    buyCondition := quantTurnedBlue and bullishCross
else
    buyCondition := (quantTurnedBlue and ema13 > ema21) or (bullishCross and isQuantBlue)

bool buyTriggered = false

if inDateRange and strategy.position_size == 0
    if buyCondition
        strategy.entry("BUY", strategy.long, comment = "BUY | Quant Biru + EMA Cross")
        buyTriggered := true

// ============================================================
// LOGIKA EXIT / SELL
// ============================================================

bool sellTriggered = false

if strategy.position_size > 0
    entryPrice      = strategy.position_avg_price
    takeProfitPrice = entryPrice * (1 + takeProfitPct / 100)
    stopLossPrice   = entryPrice * (1 - stopLossPct / 100)

    // Aturan 3: SELL saat harga CLOSE minimal -5% dari harga entry
    if close <= stopLossPrice
        strategy.close("BUY", comment = "SELL | SL -5% (Close)")
        sellTriggered := true
    else
        // Aturan 2: SELL saat profit menyentuh +5% (Limit Order running)
        strategy.exit("TP", from_entry = "BUY", limit = takeProfitPrice, comment_profit = "SELL | TP +5%")

// ============================================================
// VISUALISASI CHART
// ============================================================

plot(ema13, title = "EMA 13", color = color.green, linewidth = 2)
plot(ema21, title = "EMA 21", color = color.red, linewidth = 2)

// Garis Quant
quantSmaColor = isQuantBlue ? #0000ff : color.rgb(252, 248, 7)
plot(quantSma, title = "Quant Period SMA", color = quantSmaColor, linewidth = 2)

// Level TP (+2%) dan SL (-2% Close)
plot(strategy.position_size > 0 ? strategy.position_avg_price * (1 + takeProfitPct / 100) : na, 
     title = "TP Line (+2%)", color = color.green, style = plot.style_linebr, linewidth = 2)
plot(strategy.position_size > 0 ? strategy.position_avg_price * (1 - stopLossPct / 100) : na, 
     title = "SL Line (-2% Close)", color = color.red, style = plot.style_linebr, linewidth = 2)

// Marker Label Sinyal
plotshape(buyTriggered, title = "BUY Signal", style = shape.labelup, 
          location = location.belowbar, color = color.green, text = "BUY", textcolor = color.white, size = size.small)
plotshape(sellTriggered, title = "SELL Signal (SL)", style = shape.labeldown, 
          location = location.abovebar, color = color.red, text = "SL", textcolor = color.white, size = size.small)

// Icon Cross EMA
plotshape(bullishCross, title = "Bullish Cross (+) Icon", style = shape.cross, location = location.belowbar, color = color.green, size = size.small)
plotshape(bearishCross, title = "Bearish Cross (+) Icon", style = shape.cross, location = location.abovebar, color = color.red, size = size.small)
